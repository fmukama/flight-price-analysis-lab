# ---------------------------------------------------------------------------
# Airflow image dependencies.
#
# IMPORTANT: every version here is exact, and the Dockerfile installs this file
# under the official constraints-2.9.2 file. Both guards matter.
#
# A previous revision used `SQLAlchemy>=2.0.0`. Airflow 2.9.2 requires
# sqlalchemy>=1.4.36,<2.0, so that spec was unsatisfiable on 2.9.2 and pip
# "resolved" it by silently upgrading apache-airflow to 3.3.1 -- the only
# release allowing SQLAlchemy 2.x. Airflow 3 renamed the `airflow` console
# script, so the 2.9.2 image entrypoint (`airflow db check`) died with
# "command not found" and the scheduler/webserver crash-looped indefinitely.
# Keep the core pinned and keep SQLAlchemy below 2.0.
# ---------------------------------------------------------------------------

# Pin the core so provider/library resolution can never move it again.
apache-airflow==2.9.2

# Data manipulation and analytical computation
pandas==2.1.4
numpy==1.26.4

# Database connectors and ORM. SQLAlchemy MUST stay <2.0 for Airflow 2.9.x.
SQLAlchemy==1.4.52
pymysql==1.1.1
psycopg2-binary==2.9.9
cryptography==41.0.7

# Native Airflow database providers (versions shipped with 2.9.2)
apache-airflow-providers-mysql==5.6.1
apache-airflow-providers-postgres==5.11.1

# Parquet engine for the inter-task DataFrame handoff in data/processed/
pyarrow==14.0.2

# Test runner, so `make test` can execute the suite inside the scheduler
pytest==7.4.4
